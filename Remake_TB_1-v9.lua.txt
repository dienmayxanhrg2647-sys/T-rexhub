-- Hàm hỗ trợ chạy script an toàn, tránh crash UI nếu link chết hoặc lỗi syntax
local WindUI = nil

local function SafeLoadScript(scriptName, url, extraArg)
    task.spawn(function()
        local success, err = pcall(function()
            local source = game:HttpGet(url, true)
            local fn, loadErr = loadstring(source)
            if not fn then
                error(loadErr or "Không thể biên dịch script")
            end
            if extraArg ~= nil then
                fn(extraArg)
            else
                fn()
            end
        end)

        if not success then
            warn("[Remake TB 1] Lỗi khi chạy " .. tostring(scriptName) .. ": " .. tostring(err))
            if WindUI then
                pcall(function()
                    WindUI:Notify({
                        Title = T("ErrTitle"),
                        Content = string.format(T("ErrContent"), tostring(scriptName)),
                        Duration = 4,
                    })
                end)
            end
        end
    end)
end

local FullBrightEnabled = false
local OriginalLighting = {
    Brightness = Lighting.Brightness,
    ClockTime = Lighting.ClockTime,
    FogEnd = Lighting.FogEnd,
    GlobalShadows = Lighting.GlobalShadows,
    Ambient = Lighting.Ambient,
    OutdoorAmbient = Lighting.OutdoorAmbient,
}
local LightingConn = nil
local RespawnBrightConn = nil
local LightingAddedConn = nil

local function ApplyFullBright()
    Lighting.Brightness = 2
    Lighting.ClockTime = 14
    Lighting.FogEnd = 100000
    Lighting.GlobalShadows = false
    Lighting.Ambient = Color3.fromRGB(255, 255, 255)
    Lighting.OutdoorAmbient = Color3.fromRGB(255, 255, 255)
end

local function RestoreLighting()
    Lighting.Brightness = OriginalLighting.Brightness
    Lighting.ClockTime = OriginalLighting.ClockTime
    Lighting.FogEnd = OriginalLighting.FogEnd
    Lighting.GlobalShadows = OriginalLighting.GlobalShadows
    Lighting.Ambient = OriginalLighting.Ambient
    Lighting.OutdoorAmbient = OriginalLighting.OutdoorAmbient
end

local function EnableFullBright()
    if LightingConn then LightingConn:Disconnect() end
    if RespawnBrightConn then RespawnBrightConn:Disconnect() end
    if LightingAddedConn then LightingAddedConn:Disconnect() end

    ApplyFullBright()

    LightingConn = RunService.RenderStepped:Connect(function()
        if FullBrightEnabled then
            ApplyFullBright()
        end
    end)

    RespawnBrightConn = LocalPlayer.CharacterAdded:Connect(function()
        if FullBrightEnabled then
            task.wait(0.1)
            ApplyFullBright()
        end
    end)

    LightingAddedConn = Lighting.DescendantAdded:Connect(function(obj)
        if FullBrightEnabled and (obj:IsA("PostEffect") or obj:IsA("Atmosphere")) then
            pcall(function()
                if obj:IsA("PostEffect") then
                    obj.Enabled = false
                elseif obj:IsA("Atmosphere") then
                    obj.Density = 0
                    obj.Haze = 0
                end
            end)
        end
    end)
end

local function DisableFullBright()
    if LightingConn then LightingConn:Disconnect() LightingConn = nil end
    if RespawnBrightConn then RespawnBrightConn:Disconnect() RespawnBrightConn = nil end
    if LightingAddedConn then LightingAddedConn:Disconnect() LightingAddedConn = nil end
    RestoreLighting()
end

local FullBrightToggle = MainTab:Toggle({

        if FullBrightEnabled then

local ESPEnabled = false
local ESPObjects = {}
local ESPConn = nil
local RespawnESPConn = nil

local function RemovePlayerESP(player)
    local data = ESPObjects[player]
    if data then
        if data.Highlight then pcall(function() data.Highlight:Destroy() end) end
        if data.Billboard then pcall(function() data.Billboard:Destroy() end) end
        ESPObjects[player] = nil
    end
end

local function CreatePlayerESP(player)
    if player == LocalPlayer then return end
    RemovePlayerESP(player)

    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local head = char:FindFirstChild("Head") or hrp
    if not hrp or not head then return end

    local highlight = Instance.new("Highlight")
    highlight.Name = "TB1_ESP_Highlight"
    highlight.FillColor = Color3.fromRGB(255, 65, 65)
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.FillTransparency = 0.55
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Adornee = char
    highlight.Parent = char

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "TB1_ESP_Tag"
    billboard.Adornee = head
    billboard.Size = UDim2.new(0, 180, 0, 42)
    billboard.StudsOffset = Vector3.new(0, 2.5, 0)
    billboard.AlwaysOnTop = true

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.TextStrokeTransparency = 0.2
    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.Font = Enum.Font.GothamBold
    label.TextSize = 13
    label.Text = player.DisplayName .. " [0m]"
    label.Parent = billboard

    billboard.Parent = head

    ESPObjects[player] = {
        Highlight = highlight,
        Billboard = billboard,
        Label = label,
        HRP = hrp,
        Char = char
    }
end

local function ClearAllESP()
    for player in pairs(ESPObjects) do
        RemovePlayerESP(player)
    end
end

local function UpdateESP()
    local myChar = LocalPlayer.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if char and hrp then
                local existing = ESPObjects[player]
                if not existing
                    or existing.Char ~= char
                    or existing.HRP ~= hrp
                    or not existing.Highlight.Parent
                    or not existing.Billboard.Parent then
                    CreatePlayerESP(player)
                    existing = ESPObjects[player]
                end

                if existing and existing.Label and myHRP and myHRP.Parent then
                    local distStuds = (myHRP.Position - hrp.Position).Magnitude
                    local meters = math.floor(distStuds + 0.5)
                    existing.Label.Text = string.format("%s [%dm]", player.DisplayName, meters)
                end
            else
                RemovePlayerESP(player)
            end
        end
    end
end

Players.PlayerRemoving:Connect(function(player)
    RemovePlayerESP(player)
end)

local function EnableESP()
    if ESPConn then ESPConn:Disconnect() end
    if RespawnESPConn then RespawnESPConn:Disconnect() end

    UpdateESP()

    ESPConn = RunService.RenderStepped:Connect(function()
        if ESPEnabled then
            UpdateESP()
        end
    end)

    RespawnESPConn = LocalPlayer.CharacterAdded:Connect(function()
        if ESPEnabled then
            task.wait(0.2)
            UpdateESP()
        end
    end)
end

local function DisableESP()
    if ESPConn then ESPConn:Disconnect() ESPConn = nil end
    if RespawnESPConn then RespawnESPConn:Disconnect() RespawnESPConn = nil end
    ClearAllESP()
end

local ESPToggle = MainTab:Toggle({
    Title = T("ESPPlayer"),
    Value = false,
    Callback = function(state)
        ESPEnabled = state
        if ESPEnabled then
            EnableESP()
        else
            DisableESP()
        end
    end
})

local AutoCleanEnabled = false
local NewEffectConn = nil
local RespawnCleanConn = nil

local function IsCharacterPart(obj)
    local model = obj:FindFirstAncestorOfClass("Model")
    while model do
        if Players:GetPlayerFromCharacter(model) then
            return true
        end
        model = model:FindFirstAncestorOfClass("Model")
    end
    return false
end

local function CleanEffects(instance)
    if IsCharacterPart(instance) then return end

    if instance:IsA("ParticleEmitter")
        or instance:IsA("Trail")
        or instance:IsA("Beam")
        or instance:IsA("Fire")
        or instance:IsA("Smoke")
        or instance:IsA("Sparkles")
        or instance:IsA("Explosion") then
        instance.Enabled = false
        pcall(function() instance:Destroy() end)
    elseif instance:IsA("BloomEffect")
        or instance:IsA("BlurEffect")
        or instance:IsA("DepthOfFieldEffect")
        or instance:IsA("SunRaysEffect")
        or instance:IsA("ColorCorrectionEffect") then
        instance.Enabled = false
    elseif instance:IsA("BasePart") then
        instance.Material = Enum.Material.SmoothPlastic
        instance.Reflectance = 0
        if instance:IsA("MeshPart") then
            instance.TextureID = ""
        end
    elseif instance:IsA("Decal") or instance:IsA("Texture") then
        instance.Transparency = 1
    end
end

local function RunFixLagScan(showNotify)
    local cleanedCount = 0

    for _, obj in ipairs(Workspace:GetDescendants()) do
        local before = cleanedCount
        pcall(function()
            CleanEffects(obj)
            cleanedCount += 1
        end)
        if cleanedCount - before > 0 and cleanedCount % 250 == 0 then
            task.wait()
        end
    end

    for _, effect in ipairs(Lighting:GetChildren()) do
        if effect:IsA("PostEffect") then
            effect.Enabled = false
        end
    end

    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    end)

    if showNotify then
        WindUI:Notify({
            Title = T("FixLagTitle"),
            Content = string.format(T("FixLagContent"), tostring(cleanedCount)),
            Duration = 4,
        })
    end
end

local FixLagButton = MainTab:Button({
    Title = T("FixLag"),
    Callback = function()
        AutoCleanEnabled = true

        if not NewEffectConn then
            NewEffectConn = Workspace.DescendantAdded:Connect(function(desc)
                if AutoCleanEnabled then
                    task.defer(function()
                        pcall(function()
                            CleanEffects(desc)
                        end)
                    end)
                end
            end)
        end

        if not RespawnCleanConn then
            RespawnCleanConn = LocalPlayer.CharacterAdded:Connect(function()
                if AutoCleanEnabled then
                    task.wait(0.3)
                    pcall(function()
                        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
                    end)
                end
            end)
        end

        RunFixLagScan(true)
    end
})

local SpeedEnabled = false
local TargetSpeed = 50
local SpeedConn = nil
local RespawnSpeedConn = nil

local function StartSafeSpeed()
    if SpeedConn then SpeedConn:Disconnect() end
    if RespawnSpeedConn then RespawnSpeedConn:Disconnect() end

    -- Dịch chuyển CFrame theo MoveDirection kết hợp giữ WalkSpeed gốc và Raycast chống kẹt tường
    SpeedConn = RunService.Heartbeat:Connect(function(dt)
        if not SpeedEnabled then return end

        local char = LocalPlayer.Character
        if not char then return end

        local hum = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")

        if hum and hrp and hum.Health > 0 then
            local moveDir = hum.MoveDirection
            if moveDir.Magnitude > 0.01 then
                local baseWalk = hum.WalkSpeed > 0 and hum.WalkSpeed or 16
                local extraSpeed = math.max(0, TargetSpeed - baseWalk)

                if extraSpeed > 0 then
                    local step = moveDir.Unit * extraSpeed * math.min(dt, 0.05)

                    local rayParams = RaycastParams.new()
                    rayParams.FilterDescendantsInstances = { char }
                    rayParams.FilterType = Enum.RaycastFilterType.Exclude

                    local hit = Workspace:Raycast(hrp.Position, step + moveDir.Unit * 1.5, rayParams)
                    if not hit then
                        hrp.CFrame = hrp.CFrame + step
                    end
                end
            end
        end
    end)

    RespawnSpeedConn = LocalPlayer.CharacterAdded:Connect(function()
        if SpeedEnabled then
            task.wait(0.2)
        end
    end)
end

local function DisableSafeSpeed()
    if SpeedConn then SpeedConn:Disconnect() SpeedConn = nil end
    if RespawnSpeedConn then RespawnSpeedConn:Disconnect() RespawnSpeedConn = nil end
end

local SpeedToggle = MainTab:Toggle({
    Title = T("Speed"),
    Value = false,
    Callback = function(state)
        SpeedEnabled = state
        if SpeedEnabled then
            StartSafeSpeed()
        else
            DisableSafeSpeed()
        end
    end
})

-- CHỨC NĂNG 6: NOCLIP (TỐI ƯU CACHE PARTS & KHÔI PHỤC VA CHẠM)
-- ==========================================
local NoclipEnabled = false
local NoclipConn = nil
local NoclipCharConn = nil
local NoclipDescConn = nil
local CachedNoclipParts = {}
local OriginalCollideStates = {}

local function RebuildNoclipCache(char)
    table.clear(CachedNoclipParts)
    table.clear(OriginalCollideStates)

    if NoclipDescConn then
        NoclipDescConn:Disconnect()
        NoclipDescConn = nil
    end

    if not char then return end

    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            OriginalCollideStates[part] = part.CanCollide
            table.insert(CachedNoclipParts, part)
        end
    end

    -- Tự động cập nhật nếu có phụ kiện/vũ khí mới gắn vào nhân vật
    NoclipDescConn = char.DescendantAdded:Connect(function(desc)
        if desc:IsA("BasePart") then
            OriginalCollideStates[desc] = desc.CanCollide
            table.insert(CachedNoclipParts, desc)
        end
    end)
end

local function EnableNoclip()
    if NoclipConn then NoclipConn:Disconnect() end
    if NoclipCharConn then NoclipCharConn:Disconnect() end

    RebuildNoclipCache(LocalPlayer.Character)

    NoclipConn = RunService.Stepped:Connect(function()
        if not NoclipEnabled then return end
        for i = #CachedNoclipParts, 1, -1 do
            local part = CachedNoclipParts[i]
            if part and part.Parent then
                if part.CanCollide then
                    part.CanCollide = false
                end
            else
                table.remove(CachedNoclipParts, i)
            end
        end
    end)

    -- Tự giữ Noclip khi nhân vật chết / hồi sinh
    NoclipCharConn = LocalPlayer.CharacterAdded:Connect(function(newChar)
        if NoclipEnabled then
            task.wait(0.1)
            RebuildNoclipCache(newChar)
        end
    end)
end

local function DisableNoclip()
    if NoclipConn then NoclipConn:Disconnect() NoclipConn = nil end
    if NoclipCharConn then NoclipCharConn:Disconnect() NoclipCharConn = nil end
    if NoclipDescConn then NoclipDescConn:Disconnect() NoclipDescConn = nil end

    -- Khôi phục lại trạng thái CanCollide ban đầu cho nhân vật
    for _, part in ipairs(CachedNoclipParts) do
        if part and part.Parent then
            local orig = OriginalCollideStates[part]
            if orig ~= nil then
                part.CanCollide = orig
            end
        end
    end

    table.clear(CachedNoclipParts)
    table.clear(OriginalCollideStates)
end

local NoclipToggle = MainTab:Toggle({
    Title = T("Noclip"),
    Value = false,
    Callback = function(state)
        NoclipEnabled = state
        if NoclipEnabled then
            EnableNoclip()
        else
            DisableNoclip()
        end
    end
})

-- TAB: SETTING (CÀI ĐẶT & NGÔN NGỮ)
-- ==========================================
local SettingTab = Window:Tab({
    Title = "Setting",
    Icon = "settings",
})

local function RefreshLanguageUI()
    pcall(function()
        if FullBrightToggle and FullBrightToggle.SetTitle then FullBrightToggle:SetTitle(T("FullBright")) end
        if ESPToggle and ESPToggle.SetTitle then ESPToggle:SetTitle(T("ESPPlayer")) end
        if FixLagButton and FixLagButton.SetTitle then FixLagButton:SetTitle(T("FixLag")) end
        if FlyButton and FlyButton.SetTitle then FlyButton:SetTitle(T("Fly")) end
        if SpeedToggle and SpeedToggle.SetTitle then SpeedToggle:SetTitle(T("Speed")) end
        if SpeedSlider and SpeedSlider.SetTitle then SpeedSlider:SetTitle(T("SpeedValue")) end
        if NoclipToggle and NoclipToggle.SetTitle then NoclipToggle:SetTitle(T("Noclip")) end
        if ScriptCollectionSection and ScriptCollectionSection.SetTitle then
            ScriptCollectionSection:SetTitle(T("ScriptCollection"))
        end
    end)
end

-- Chức năng 1: Language (4 ngôn ngữ: Anh, Việt, Nga, Pháp)
SettingTab:Dropdown({
    Title = "Language",
    Desc = "English / Tiếng Việt / Русский / Français",
    Values = { "English", "Tiếng Việt", "Русский", "Français" },
    Value = CurrentLanguage,
    Callback = function(selected)
        CurrentLanguage = selected
        RefreshLanguageUI()
        WindUI:Notify({
            Title = T("LangChangedTitle"),
            Content = T("LangChangedContent"),
            Duration = 3,
        })
    end
})
